Hebrew collation test (I18N-07) — needs at least 3 soldier rows with different Hebrew first letters to verify sorting order: e.g., אבי (aleph), בעז (bet), גדעון (gimel). Codepoint sort would put U+05D0..05D2 in order — same as alphabetic. To prove ICU is doing something different from codepoint, need names where codepoint and Hebrew alphabetic order DIFFER. Tricky in Hebrew because Hebrew letters are codepoint-ordered. Workaround: include a name with a final-form letter (e.g., מ vs ם — final mem U+05DD is codepoint-ordered AFTER mem U+05DE but in Hebrew alphabetic order they're treated as the same letter). Use names: שלום (with final mem U+05DD), שלם (without final form), אבי. Expected ICU order: אבי, שלום, שלם. Codepoint order would put שלום before שלם (final mem comes BEFORE regular mem in codepoint). Tradeoff: this is subtle. Alternative simpler test: just verify the collation exists and that ORDER BY uses it (check via `EXPLAIN`).

Pragmatic test: insert two soldiers, one named שלום and one named שלם. ORDER BY display_name. Assert that ICU collation returns them in alphabetic Hebrew order (whatever that is — query the result and compare against a fixed expected order). If the test is too brittle, just verify pg_collation contains 'he-x-icu' and the column was created with that collation (via `\d+ soldier`).

    import { Client } from 'pg';
    import { randomUUID, randomBytes } from 'node:crypto';

    const PG_URL = process.env.PG_TEST_URL ?? 'postgres://shifts:changeme@localhost:5432/shifts';

    export interface TenantFixture {
      tenantId: string;
      orgUnitId: string;
      adminEmail: string;
      adminUserId: string;   // app_user.id
      adminAuthUserId: string; // users.id (Auth.js)
      adminSoldierId: string;
      inviteCode: string;    // 8-char Crockford
      inviteCodeId: string;
    }

    function crockford8(): string {
      const alphabet = '0123456789ABCDEFGHJKMNPQRSTVWXYZ';
      const bytes = randomBytes(8);
      let out = '';
      for (let i = 0; i < 8; i++) out += alphabet[bytes[i] % 32];
      return out;
    }

    async function seedOne(client: Client, label: 'A' | 'B'): Promise<TenantFixture> {
      // BYPASS RLS for seeding — we run as superuser-equivalent in tests OR set app.current_tenant per insert.
      // Simpler: set app.current_tenant before each tenant's inserts.
      const tenantId = randomUUID();
      const orgUnitId = randomUUID();
      const adminEmail = `admin-${label.toLowerCase()}@example.test`;
      const adminAuthUserId = randomUUID(); // users.id (Auth.js)
      const adminUserId = randomUUID();     // app_user.id
      const adminSoldierId = randomUUID();
      const inviteCodeId = randomUUID();
      const inviteCode = crockford8();

      // tenant.id is the wildcard — but RLS on tenant uses USING (id = current_setting('app.current_tenant')::uuid)
      // For the INITIAL tenant insert, app.current_tenant must equal the new tenant's id, OR we use a security definer / superuser.
      // Pragmatic: set app.current_tenant to tenantId before tenant insert.
      await client.query(`SELECT set_config('app.current_tenant', $1, true)`, [tenantId]);
      await client.query(
        `INSERT INTO tenant (id, name, org_depth) VALUES ($1, $2, 1)`,
        [tenantId, `Test Tenant ${label}`]
      );
      await client.query(
        `INSERT INTO org_unit (id, tenant_id, parent_id, level, name) VALUES ($1, $2, NULL, 1, $3)`,
        [orgUnitId, tenantId, `Test Tenant ${label}`]
      );
      // Insert Auth.js users row first (so app_user.user_id FK resolves)
      await client.query(
        `INSERT INTO "users" (id, name, email, "emailVerified") VALUES ($1, $2, $3, now())`,
        [adminAuthUserId, `admin-${label}`, adminEmail]
      );
      await client.query(
        `INSERT INTO app_user (id, tenant_id, email, display_name, locale, user_id) VALUES ($1, $2, $3, $4, 'he', $5)`,
        [adminUserId, tenantId, adminEmail, `Admin ${label}`, adminAuthUserId]
      );
      await client.query(
        `INSERT INTO soldier (id, tenant_id, user_id, display_name) VALUES ($1, $2, $3, $4)`,
        [adminSoldierId, tenantId, adminUserId, `Admin ${label} Soldier`]
      );
      await client.query(
        `INSERT INTO membership (id, tenant_id, soldier_id, org_unit_id, role) VALUES ($1, $2, $3, $4, 'unit_admin')`,
        [randomUUID(), tenantId, adminSoldierId, orgUnitId]
      );
      await client.query(
        `INSERT INTO invite_code (id, tenant_id, code, org_unit_id, role, created_by) VALUES ($1, $2, $3, $4, 'member', $5)`,
        [inviteCodeId, tenantId, inviteCode, orgUnitId, adminUserId]
      );

      return { tenantId, orgUnitId, adminEmail, adminUserId, adminAuthUserId, adminSoldierId, inviteCode, inviteCodeId };

    export async function seedTwoTenants(): Promise<{ tenantA: TenantFixture; tenantB: TenantFixture }> {
      const client = new Client({ connectionString: PG_URL });
      await client.connect();

        const tenantA = await seedOne(client, 'A');
        const tenantB = await seedOne(client, 'B');
        return { tenantA, tenantB };
      } finally {
        await client.end();
      }
    }

    export interface SignInResult {
      sessionToken: string;
      userId: string;       // Auth.js users.id
      cookies: string;      // Cookie header value for Playwright request context
    }

    /**
     * Inserts a session row directly into the `sessions` table (bypassing the email link click).
     * Returns the cookie value that NextAuth would set after a successful magic-link callback.
     * Plan 04 uses this for speed; Plan 03 Task 7 exercised the real email flow.
     */
    export async function signInAs(email: string): Promise<SignInResult> {
      const client = new Client({ connectionString: PG_URL });
      await client.connect();
      try {
        const userResult = await client.query<{ id: string }>(
          `SELECT id FROM "users" WHERE email = $1`,
          [email]
        );
        if (userResult.rows.length === 0) {
          throw new Error(`signInAs: no users row for email ${email}`);
        }
        const userId = userResult.rows[0].id;
        const sessionToken = randomBytes(32).toString('hex');
        await client.query(
          `INSERT INTO sessions (id, "sessionToken", "userId", expires) VALUES ($1, $2, $3, now() + interval '30 days')`,
          [randomUUID(), sessionToken, userId]
        );
        // NextAuth default cookie name: `next-auth.session-token` (or `__Secure-next-auth.session-token` in production)
        // Test runs typically use HTTP not HTTPS — cookie name without __Secure prefix
        const cookies = `next-auth.session-token=${sessionToken}; Path=/; HttpOnly`;
        return { sessionToken, userId, cookies };
      } finally {
        await client.end();
      }
    }
