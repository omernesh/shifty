    import { test } from 'node:test';
    import assert from 'node:assert/strict';
    import { canonicalizeText } from '../../app/plugins/shifty-roster/src/helpers/canonicalize.js';

    test('canonicalize: kibbutz canary — U+2019 stripped', () => {
      // נועם ג’לאל (with U+2019 right single quotation mark)
      // → נועם גלאל (no apostrophe at all — U+2019 stripped to empty)
      assert.equal(canonicalizeText('נועם ג’לאל'), 'נועם גלאל');
    });

    test('canonicalize: bidi marks stripped (U+200E, U+200F, U+202A–E)', () => {
      for (const cp of [0x200E, 0x200F, 0x202A, 0x202B, 0x202C, 0x202D, 0x202E]) {
        const mid = 'ABC' + String.fromCodePoint(cp) + 'DEF';
        const out = canonicalizeText(mid);
        assert.equal(out.indexOf(String.fromCodePoint(cp)), -1, `codepoint U+${cp.toString(16)} not stripped`);
      }
    });

    test('canonicalize: Hebrew gershayim U+05F4 preserved', () => {
      assert.equal(canonicalizeText('סמ״ר דני'), 'סמ״ר דני');
    });

    test('canonicalize: ASCII apostrophe preserved', () => {
      assert.equal(canonicalizeText("D'Angelo"), "D'Angelo");
    });

    test('canonicalize: null/undefined to empty string', () => {
      assert.equal(canonicalizeText(null), '');
      assert.equal(canonicalizeText(undefined), '');
    });
