  SELECT s.id AS soldier_id, s.display_name,
    count(av.id) FILTER (WHERE av.declared = 'unavailable' OR av.source = 'manager_override') AS declared_count,
    (SELECT count(*) FROM shift_instance si WHERE si.planning_window_id = :pw_id AND si.tenant_id = :tenant_id) AS total_instances
