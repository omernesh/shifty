RESEARCH Recipe 4 OpenPlanningWindow contract (this plan's load-bearing handler):
  Payload: { team_id, start_date, end_date, constraint_lock_at }  (lock nullable → default)
  Returns: { planning_window_id, instance_count }
  Throws on: missing team_id/dates, end<start, days>30, team has zero slots, instance_count>3600

INSERT…SELECT cross-product SQL (verbatim from RESEARCH Recipe 4):
  INSERT INTO shift_instance (tenant_id, shift_slot_id, planning_window_id, date, headcount_index)
  SELECT s.tenant_id, s.id, :pw_id::uuid, d.date::date, h.idx
  FROM shift_slot s
  CROSS JOIN generate_series(:start_date::date, :end_date::date, INTERVAL '1 day') AS d(date)
  CROSS JOIN LATERAL generate_series(0, s.headcount - 1) AS h(idx)
  WHERE s.tenant_id = :tenant_id AND s.team_id = :team_id
  RETURNING id

constraint_lock_at default expression (RESEARCH Recipe 4 + Pitfall 7):
  (:start_date::date - INTERVAL '3 days')::date + TIME '23:59:00' AT TIME ZONE 'Asia/Jerusalem'

planning_window schema (0003):
  id UUID PK, tenant_id UUID, team_id UUID FK, start_date DATE, end_date DATE,
  constraint_lock_at TIMESTAMPTZ, state TEXT CHECK in (open|draft|published|closed),
  created_at, updated_at
  + CHECK end_date >= start_date

shift_instance schema (0003 + 0014 added planning_window_id NOT NULL):
  id UUID PK, tenant_id UUID, shift_slot_id UUID FK, planning_window_id UUID FK,
  date DATE, headcount_index INT, UNIQUE(shift_slot_id, date, headcount_index)

availability FK chain post-0014:
  availability.shift_instance_id → shift_instance.id  (ON DELETE CASCADE)
