Purpose: Resolves BUDIBASE-CONVENTIONS.md §10 open item #1. The 4 helpers + 26 unit tests are the most reusable artifact from the Lowdefy era. Their semantics (smart-quote stripping, 24-color round-robin, role-tag kebab-case, availability source precedence) are load-bearing for the Phase 3 W1+ availability flows. Bundling them into a single IIFE that any Builder UI JS code block can paste-and-call removes the friction of re-implementing these primitives per-Automation.

Sequencing rationale (per D-06):
- W0-03 can run in parallel with W0-01 (W0-01 is doc-only PRD edit).
- W0-02 depends on this plan (BUDIBASE-CONVENTIONS.md §3 procedure for invite-redemption Automation may use a helper; even if it doesn't, validating the bundling pattern before the first Automation ships is the right ordering).

Discretion (per D-01): esbuild flags choice. Decision (recorded here for executor; final flags TBD by executor based on build output):
- `--bundle --format=iife --global-name=Shifty` (required by D-01)
- `--platform=neutral` (Budibase JS sandbox is browser-shaped, no Node.js APIs)
- `--target=es2020` (Budibase's JS sandbox is modern enough; ES2020 unlocks optional chaining, nullish coalescing, BigInt — none load-bearing for these helpers but harmless)
- `--minify` (smaller bundle = faster paste into Builder UI; sourcemap as a sibling .map for debug)
- `--sourcemap=external` (debugging in Builder UI sandbox is painful without; external keeps the bundle file small)

Builder UI consumption pattern (decision in this plan, documented in README):
- Primary pattern: **paste-as-fixture**. The bundle source is pasted as the first ~50 lines of any Builder UI JS code block that needs a helper. Pattern: `// === BEGIN Shifty helpers bundle ===\n<bundle content>\n// === END Shifty helpers bundle ===\n\n// Your code below uses Shifty.canonicalizeText(...) etc.`
- Why this and not HTTP fetch: Budibase JS sandbox has limited network capability for arbitrary CDNs; fetching helpers per-Automation would add latency and a network-dep failure mode. Paste-as-fixture trades bundle size (~5KB minified) for reliability.
- Rejected: NPM-package-in-snapshot-tarball (per D-01 already-rejected list). Rejected: inline-copy-per-Automation (also already-rejected; duplication explosion).

    // tools/budibase-helpers/build.mjs
    // Bundles the 4 pure-function helpers into a single IIFE exposing global `Shifty`.
    // Output: helpers.bundle.js + helpers.bundle.js.map (external sourcemap).
    // Consumed by Builder UI JS code blocks via paste-as-fixture (see README.md).
    import { build } from 'esbuild';
    import { writeFileSync } from 'fs';
    import { fileURLToPath } from 'url';
    import { dirname, join } from 'path';

    const __dirname = dirname(fileURLToPath(import.meta.url));

    // The IIFE entry file aggregates the 4 helpers and exports them as a single object.
    const entryContent = `
    export { canonicalizeText } from './src/canonicalize.js';
    export { PALETTE, pickNextColor, colorByIndex } from './src/palette.js';
    export { canonicalizeRoleTag } from './src/role-tag.js';
    export { SOURCE_RANK, SOURCE_VALUES } from './src/availability-source.js';
    `;
    const entryPath = join(__dirname, '.entry.mjs');
    writeFileSync(entryPath, entryContent, 'utf8');

    await build({
      entryPoints: [entryPath],
      outfile: join(__dirname, 'helpers.bundle.js'),
      bundle: true,
      format: 'iife',
      globalName: 'Shifty',
      platform: 'neutral',
      target: 'es2020',
      minify: true,
      sourcemap: 'external',
      logLevel: 'info',
    });

    // esbuild's globalName produces `var Shifty = (() => { ... return { canonicalizeText, ... } })();`
    // — which is exactly the API we want. Verify by reading helpers.bundle.js after build.
    console.log('Bundle built: helpers.bundle.js + helpers.bundle.js.map');
    `;

    // tests/unit/bundle-shifty-global.spec.ts
    // Verifies the IIFE bundle at tools/budibase-helpers/helpers.bundle.js
    // exposes the global `Shifty` with all expected helper exports.
    // This is the contract surface for Builder UI JS code blocks consuming
    // the paste-as-fixture pattern documented in tools/budibase-helpers/README.md.
    import { test } from 'node:test';
    import assert from 'node:assert/strict';
    import { readFileSync } from 'node:fs';
    import { fileURLToPath } from 'node:url';
    import { dirname, join } from 'node:path';

    const __dirname = dirname(fileURLToPath(import.meta.url));
    const BUNDLE_PATH = join(__dirname, '..', '..', 'tools', 'budibase-helpers', 'helpers.bundle.js');
    const bundleCode = readFileSync(BUNDLE_PATH, 'utf8');

    // Evaluate the bundle in a controlled scope. The IIFE produces `var Shifty = (...);`
    // at the top of the bundle, so eval makes Shifty available in this scope.
    // Use indirect eval (window-eval semantics) to keep strict mode out of the way.
    const indirectEval = eval;
    let Shifty: any;
    indirectEval(`Shifty = (function(){ ${bundleCode}; return Shifty; })();`);

    test('Shifty.canonicalizeText strips U+2019 (smart quote)', () => {
      assert.equal(Shifty.canonicalizeText('נועם ג’לאל'), 'נועם גלאל');
    });

    test('Shifty.PALETTE is the 24-color frozen array', () => {
      assert.equal(Shifty.PALETTE.length, 24);
      assert.equal(Shifty.PALETTE[0], '#1F77B4');
      assert.equal(Shifty.PALETTE[23], '#843C39');
    });

    test('Shifty.pickNextColor handles null/undefined/negative sentinel', () => {
      assert.equal(Shifty.pickNextColor(null), 0);
      assert.equal(Shifty.pickNextColor(undefined), 0);
      assert.equal(Shifty.pickNextColor(-1), 0);
      assert.equal(Shifty.pickNextColor(0), 2);
      assert.equal(Shifty.pickNextColor(22), 0); // wraps
    });

    test('Shifty.colorByIndex out-of-range falls back to PALETTE[0]', () => {
      assert.equal(Shifty.colorByIndex(0), '#1F77B4');
      assert.equal(Shifty.colorByIndex(null), '#1F77B4');
      assert.equal(Shifty.colorByIndex(99), '#1F77B4');
    });

    test('Shifty.canonicalizeRoleTag lowercases + kebabs + NFC-normalizes', () => {
      assert.equal(Shifty.canonicalizeRoleTag('Medic Officer'), 'medic-officer');
      assert.equal(Shifty.canonicalizeRoleTag('medic’s'), 'medics');
      assert.equal(Shifty.canonicalizeRoleTag(''), '');
      assert.equal(Shifty.canonicalizeRoleTag(null), '');
    });

    });
